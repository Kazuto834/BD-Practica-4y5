--- 2. UPDATE CON FROM Y CASE ---
UPDATE UBICACION u
SET Capacidad_Peso_Kg = u.Capacidad_Peso_Kg - sub.Peso_Total
FROM (
    SELECT inv.ID_Ubicacion, SUM(inv.Cantidad * p.Peso_Kg) AS Peso_Total
    FROM INVENTARIO_FISICO inv JOIN PRODUCTO p ON inv.SKU = p.SKU GROUP BY inv.ID_Ubicacion
) sub
WHERE u.ID_Ubicacion = sub.ID_Ubicacion AND (u.Capacidad_Peso_Kg - sub.Peso_Total) > 0;
UPDATE 60

    ELSE Estatus
END
WHERE Estatus IN ('Pendiente', 'En picking');
